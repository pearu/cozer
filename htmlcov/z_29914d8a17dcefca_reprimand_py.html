1"""Reprimand notice (portrait, one page) — a blank fill-in form for a U.I.M. §406.04 reprimand. 

2 

3A reprimand is a formal notice of disapproval of an unacceptable action (U.I.M. Circuit Rules 2026 

4§406.04): it must be recorded by the Race Secretariat and witnessed, it constitutes a warning that a 

5recurrence draws a heavier penalty, and it is valid 12 months. This report prints the notice as a 

6STATIC form the O.O.D. completes by hand for a specific incident — event header + the O.O.D./Secretary 

7signature lines are filled from the event; the competitor and the offence are write-in. English only 

8(U.I.M. official-form style; the article numbers are universal), matching the Inspection forms. 

9""" 

10from cozer.reports.common import esc, meta_of, document_html 

11from cozer.reports.labels import get_labels 

12from cozer.reports.render import render_pdf 

13 

14# The four §406.03 "unacceptable behaviour" grounds; a fifth "other breach" line covers a rules breach. 

15_GROUNDS = ("a deliberate act to gain unfair advantage", 

16 "a false act or statement suppressing facts required for the proper conduct of the race", 

17 "an attempt to bribe, or the taking of a bribe", 

18 "abusive or unsportsmanlike behaviour") 

19 

20_CSS = """ 

21.rn-note { margin:4px 0 10px; font-size:10.5px; color:#333; } 

22.rn-sec { font-weight:bold; margin:12px 0 4px; font-size:12px; 

23 border-bottom:2px solid #222; padding-bottom:2px; } 

24.rn-field { display:flex; align-items:flex-end; gap:6px; margin:6px 0; font-size:11px; } 

25.rn-lbl { white-space:nowrap; } 

26.rn-line { flex:1 1 auto; border-bottom:1px solid #555; min-height:1.15em; } 

27.rn-grid { display:flex; gap:14px; } 

28.rn-grid .rn-field { flex:1; } 

29.rn-box { display:inline-block; width:11px; height:11px; border:1.3px solid #333; 

30 margin-right:6px; vertical-align:-1px; } 

31.rn-check { font-size:11px; margin:3px 0; } 

32.rn-write { min-height:15.5em; border:1px solid #999; margin-top:3px; } 

33.rn-statement { border:2px solid #222; padding:9px 11px; margin:12px 0; font-size:11px; 

34 line-height:1.45; background:#f7f7f7; } 

35.rn-statement b { text-transform:uppercase; } 

36.rn-rights { font-size:9.5px; color:#444; margin:6px 0 2px; }

38.rn-sig .rn-cell { flex:1; } 

39.rn-sig .rn-sigline { border-bottom:1px solid #333; height:1.9em; } 

40.rn-sig .rn-cap { font-size:9px; color:#444; margin-top:2px; } 

41.rn-name { font-size:11px; font-weight:bold; } 

42"""

45def _field(label): 

46 return ('<div class="rn-field"><span class="rn-lbl">%s</span><span class="rn-line"></span></div>' 

47 % esc(label)) 

48 

49 

50def _sig(name, caption): 

51 return ('<div class="rn-cell"><div class="rn-sigline"></div><div class="rn-cap">%s</div>%s</div>' 

52 % (esc(caption), ('<div class="rn-name">%s</div>' % esc(name)) if name else "")) 

53 

54 

55def _body(meta): 

56 ood = (meta.get("officer") or "").strip() 

57 sec = (meta.get("secretary") or "").strip() 

58 b = ['<style>%s</style>' % _CSS] 

59 b.append('<div class="rn-note">Issued under Article <b>406.04</b> of the U.I.M. Circuit Racing Rules ' 

60 '2026. A reprimand is a formal notice of disapproval of an unacceptable action; it must be ' 

61 'recorded by the Race Secretariat and witnessed, and it constitutes a warning that a ' 

62 'recurrence will draw a heavier penalty (valid 12 months).</div>') 

63 

64 b.append('<div class="rn-sec">Competitor</div>') 

65 b.append('<div class="rn-grid">%s%s</div>' % (_field("Driver / competitor"), _field("Boat No."))) 

66 b.append('<div class="rn-grid">%s%s%s</div>' 

67 % (_field("Class"), _field("Nationality"), _field("Licence No."))) 

68 

69 b.append('<div class="rn-sec">The unacceptable action</div>') 

70 b.append('<div class="rn-grid">%s%s</div>' % (_field("Date and time of the incident"), 

71 _field("Race / heat / session"))) 

72 b.append('<div class="rn-field"><span class="rn-lbl">Description</span></div>') 

73 b.append('<div class="rn-write"></div>') 

74 b.append('<div class="rn-field" style="margin-top:8px"><span class="rn-lbl">Ground ' 

75 '(&sect;406.03 &mdash; tick as applicable):</span></div>') 

76 for g in _GROUNDS: 

77 b.append('<div class="rn-check"><span class="rn-box"></span>%s</div>' % esc(g)) 

78 b.append('<div class="rn-check"><span class="rn-box"></span>other breach of the applicable rules: ' 

79 '<span class="rn-line" style="display:inline-block;width:60%;"></span></div>') 

80 

81 b.append('<div class="rn-statement">You are hereby issued a <b>reprimand</b> &mdash; a formal notice ' 

82 'of disapproval of the unacceptable action described above, under U.I.M. Circuit Rules ' 

83 '&sect;406.04. This reprimand <b>automatically constitutes a warning</b>: should the offence ' 

84 'recur, a heavier penalty will be imposed. It is <b>valid for twelve (12) months</b> from the ' 

85 'date of issue below and is <b>recorded by the Race Secretariat</b>.</div>') 

86 b.append('<div class="rn-rights">The competitor may lodge a protest against this penalty as provided ' 

87 'in &sect;403, and, if it is not upheld, may appeal under &sect;405.</div>') 

88 

89 b.append('<div class="rn-sec">Issue, witness and record</div>') 

90 b.append('<div class="rn-grid">%s%s</div>' % (_field("Date issued"), _field("Time issued"))) 

91 b.append('<div class="rn-sig">%s%s</div>' 

92 % (_sig(ood, "Officer of the Day / Race Director — signature (§406.04)"), 

93 _sig("", "Witness — name, role and signature (§406.04)"))) 

94 b.append('<div class="rn-sig" style="margin-top:14px">%s%s</div>' 

95 % (_sig(sec, "Recorded by the Secretary of the Race — signature (§406.04)"), 

96 _sig("", "Received by the competitor — signature (refusal to sign is noted)"))) 

97 return b

100def build_reprimand(eventdata, classes=None): 

101 return {"meta": meta_of(eventdata), "labels": get_labels(eventdata), "orientation": "portrait", 

102 "heading": "Reprimand", "subtitle": "U.I.M. Circuit Rules §406.04"} 

103 

104 

105def reprimand_html(model): 

106 return document_html(model["orientation"], model["labels"], model["meta"], model["heading"], 

107 _body(model["meta"]), subtitle=model["subtitle"], posting=False) 

108 

109 

110def render_reprimand(eventdata, out_path, classes=None): 

111 render_pdf(reprimand_html(build_reprimand(eventdata)), out_path)
