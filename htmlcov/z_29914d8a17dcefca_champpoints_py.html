1"""Championship Points export (issue #62). 

2 

3A cross-class **spreadsheet** of each driver's FINAL event ranking + points, for feeding a multi-event 

4championship / series where this event is one round. Unlike the other reports this is a CSV (stdlib, opens 

5in any spreadsheet program — no new dependency), not a PDF. 

6 

7**Model A** (the shipped scope): the RANK is the authoritative, series-independent export; the event's own 

8points ride along only as a *reference* column, because the series may use a **different scoring system** 

9than the race event and applies its own table downstream. So cozer never presents the event points AS 

10championship points. 

11 

12Reuses the Full-Final standings assembly (``build_full_final`` → ``sumanalyze``/``getsumresorder``) — no 

13core-scoring change; the ranking is exactly what the Full Final report shows. 

14""" 

15import csv 

16import io 

17 

18from cozer.racepattern import get_classes 

19from cozer.reports.common import participants_index, nationalities_index 

20from cozer.reports.final import build_full_final 

21 

22# The spreadsheet columns, in order. Event/Date identify the round so a championship tally can concatenate 

23# several rounds' exports; Place is the authoritative input, Points the event's own (reference) score. 

24COLUMNS = ["Event", "Date", "Class", "Place", "Boat", "First name", "Last name", 

25 "Club", "Nationality", "Points"] 

26 

27 

28def _event_name(eventdata): 

29 return (eventdata.get("broadcast", {}) or {}).get("eventname") or eventdata.get("title", "") or "" 

30 

31 

32def _race_classes(eventdata): 

33 """The RACE/final classes — the base class names, without the ``/T`` (time-trial) or ``/Q`` 

34 (qualification) phase variants. Championship points come from the final race result, never the 

35 time trial (which scores 0), so those variants are excluded.""" 

36 return [cl for cl in get_classes(eventdata) if "/" not in cl] 

37 

38 

39def build_championship_points(eventdata, classes=None): 

40 """Rows (list of dicts keyed by ``COLUMNS``) for the selected classes' final RACE standings. 

41 

42 ``classes`` is the Reports-tab class selection (``None`` = all classes); any phase variant is 

43 normalized to its base race class, so the export always reflects the final race result (not the time 

44 trial). A driver who started but never classified appears with a blank Place and Points. Ordering: 

45 classes as selected/declared, drivers by final rank (unclassified last).""" 

46 race = _race_classes(eventdata) 

47 if classes is None: 

48 targets = race 

49 else: # normalize F 500/T -> F 500, keep order, dedupe 

50 seen, targets = set(), [] 

51 for cl in classes: 

52 base = cl.split("/")[0] 

53 if base in race and base not in seen: 

54 seen.add(base) 

55 targets.append(base) 

56 model = build_full_final(eventdata, classes=targets) 

57 parts = participants_index(eventdata) 

58 nats = nationalities_index(eventdata) 

59 event, date = _event_name(eventdata), eventdata.get("date", "") or "" 

60 rows = [] 

61 for tbl in model["tables"]: 

62 cl = tbl["class"] # display name == participant-index base key 

63 for r in tbl["rows"]: 

64 pid = str(r["id"]) 

65 first, last, club = parts.get((cl, pid), ("", "", "")) 

66 points = r["sumpoints"] 

67 rows.append({ 

68 "Event": event, "Date": date, "Class": cl, 

69 "Place": r["place"], # "" when unclassified/DNQ 

70 "Boat": pid, "First name": first, "Last name": last, 

71 "Club": club, "Nationality": nats.get((cl, pid), "") or r.get("nat", ""), 

72 "Points": "" if points in ("-", None) else points, # reference only (see module doc) 

73 }) 

74 return rows 

75 

76 

77def championship_points_csv(rows): 

78 """``rows`` as CSV text: the ``COLUMNS`` header then one line per driver.""" 

79 buf = io.StringIO() 

80 w = csv.DictWriter(buf, fieldnames=COLUMNS, extrasaction="ignore") 

81 w.writeheader() 

82 w.writerows(rows) 

83 return buf.getvalue() 

84 

85 

86def render_championship_points(eventdata, path, classes=None): 

87 """Write the Championship Points CSV to ``path``; returns the rows.""" 

88 rows = build_championship_points(eventdata, classes=classes) 

89 with open(path, "w", newline="", encoding="utf-8") as f: 

90 f.write(championship_points_csv(rows)) 

91 return rows
